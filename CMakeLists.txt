cmake_minimum_required(VERSION 3.24)
project(IPCRevival VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This milestone supports Windows MSVC x64 only")
endif()
set(IPC_BOOST_ROOT "" CACHE PATH "Extracted Boost 1.92.0 distribution (no download during configure)")
if(NOT EXISTS "${IPC_BOOST_ROOT}/boost/version.hpp")
  message(FATAL_ERROR "Set IPC_BOOST_ROOT to a Boost 1.92.0 distribution")
endif()
file(STRINGS "${IPC_BOOST_ROOT}/boost/version.hpp" BOOST_VERSION_LINE REGEX "^#define BOOST_VERSION [0-9]+$")
if(NOT BOOST_VERSION_LINE STREQUAL "#define BOOST_VERSION 109200")
  message(FATAL_ERROR "This milestone pins Boost 1.92.0 for shared-memory ABI consistency")
endif()
file(GLOB IPC_NATIVE_SOURCES CONFIGURE_DEPENDS Native/Src/*.cpp Native/Src/detail/*.cpp Native/Src/Policies/*.cpp)
add_library(ipc_native STATIC ${IPC_NATIVE_SOURCES})
add_library(IPC::Native ALIAS ipc_native)
target_compile_features(ipc_native PUBLIC cxx_std_17)
target_include_directories(ipc_native PUBLIC Inc PRIVATE Native/Inc)
target_include_directories(ipc_native SYSTEM PUBLIC "${IPC_BOOST_ROOT}")
target_compile_definitions(ipc_native PUBLIC BOOST_USE_WINDOWS_H BOOST_ALL_NO_LIB NOMINMAX _WIN32_WINNT=0x0A00 PRIVATE _CRT_SECURE_NO_WARNINGS _SCL_SECURE_NO_WARNINGS)
target_compile_options(ipc_native PRIVATE /W4 /permissive- /EHsc /utf-8)
add_executable(ipc_calc Examples/Calc/CalcNative/main.cpp Examples/Calc/CalcNative/Client.cpp Examples/Calc/CalcNative/Server.cpp)
target_link_libraries(ipc_calc PRIVATE IPC::Native)
target_compile_options(ipc_calc PRIVATE /W4 /permissive- /EHsc /utf-8)
include(CTest)
add_executable(ipc_worker Examples/worker.cpp)
target_link_libraries(ipc_worker PRIVATE IPC::Native)
target_compile_options(ipc_worker PRIVATE /W4 /permissive- /EHsc /utf-8)
if(BUILD_TESTING)
  add_executable(ipc_core_tests tests/core_tests.cpp)
  target_link_libraries(ipc_core_tests PRIVATE IPC::Native)
  target_compile_options(ipc_core_tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  add_test(NAME core_regressions COMMAND ipc_core_tests)
  set_tests_properties(core_regressions PROPERTIES TIMEOUT 15)
  find_program(IPC_PWSH pwsh REQUIRED)
  add_test(NAME process_integration COMMAND "${IPC_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/process_tests.ps1" -Executable $<TARGET_FILE:ipc_worker> -Results "${CMAKE_CURRENT_BINARY_DIR}/process-results-$<CONFIG>.json")
  set_tests_properties(process_integration PROPERTIES TIMEOUT 120)
endif()
